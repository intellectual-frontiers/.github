% A glyph the current font lacks falls back to STIX Two Math, then Inter, and never vanishes (spec FR-020). XeLaTeX has no
% font fallback, so each symbol a family in use can lack is an active character that checks the current font first.

\font\ifsymprobea="[\iffontdir STIXTwoMath-Regular.otf]" at 10pt
\font\ifsymprobeb="[\iffontdir Inter-Regular.ttf]" at 10pt
\newfontfamily\ifsymfonta{STIXTwoMath-Regular.otf}[Path=\iffontdir]
\newfontfamily\ifsymfontb{Inter-Regular.ttf}[Path=\iffontdir]
\newcommand{\ifsymfallback}[1]{%
  \iffontchar\font"#1 \char"#1 \else
    \iffontchar\ifsymprobea"#1 {\ifsymfonta\char"#1 }\else {\ifsymfontb\char"#1 }\fi
  \fi}
